    requestAnimationFrame(animate);
    t += 0.012;

    camera.position.x += (mouseX * 3 - camera.position.x) * 0.05;
    camera.position.y += (-mouseY * 2 - camera.position.y) * 0.05;
    camera.position.z = 20 - scrollY * 0.008;
    camera.lookAt(0, 0, 0);

    // Animate salon items and pearls
    objects.forEach(obj => {
      const d = obj.userData;
      obj.rotation.x += d.rotX;
      obj.rotation.y += d.rotY;
      obj.position.y = d.baseY + Math.sin(t * d.floatSpeed + d.phase) * d.floatAmp * 4;
    });

    // Animate Dust Particles
    const positions = stars.geometry.attributes.position.array;
    for(let i=0; i<starCount; i++) {
      positions[i*3+1] += starVel[i];
      if(positions[i*3+1] > 30) {
        positions[i*3+1] = -30; 
      }
    }
    stars.geometry.attributes.position.needsUpdate = true;

    // Pulse lights dynamically
    pinkLight.intensity = 2.5 + Math.sin(t * 1.5) * 1;
    goldLight.intensity = 2.0 + Math.cos(t * 1.2) * 0.8;

    renderer.render(scene, camera);
  }
  
  window.onload = function() {
      animate();
  };
})();

  entries.forEach((e, i) => {

    }
  });

/* ============================================================
   COUNTER ANIMATION
   ============================================================ */
const counterObs = new IntersectionObserver(entries => {
  entries.forEach(entry => {
    if (entry.isIntersecting) {
      const nums = entry.target.querySelectorAll('[data-target]');
      nums.forEach(el => {
        const target = +el.dataset.target;
        let current = 0;
        const step = target / 60;
        const timer = setInterval(() => {
          current += step;
          if (current >= target) { current = target; clearInterval(timer); }
          el.textContent = Math.floor(current) + (target === 100 ? '%' : '+');
        }, 25);
      });
      counterObs.unobserve(entry.target);
    }
  });
}, { threshold: 0.5 });
const statsEl = document.querySelector('.stats-strip');
if (statsEl) counterObs.observe(statsEl);

/* ============================================================
   3D TILT ON CARDS
   ============================================================ */
document.querySelectorAll('.service-card, .team-card, .feature-box').forEach(card => {
  card.addEventListener('mousemove', e => {
    const rect = card.getBoundingClientRect();
    const cx = rect.left + rect.width / 2;
    const cy = rect.top + rect.height / 2;
    const rx = (e.clientY - cy) / (rect.height / 2) * -6;
    const ry = (e.clientX - cx) / (rect.width / 2) * 6;
    card.style.transform = `perspective(800px) rotateX(${rx}deg) rotateY(${ry}deg) translateY(-5px)`;
  });
  card.addEventListener('mouseleave', () => {
    card.style.transform = '';
  });
});
